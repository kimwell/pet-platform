async function login(domain,body){const pre=await read(domain,'csrf');const r=await send(domain,'login',{method:'POST',headers:{'Content-Type':'application/json','X-CSRF-Token':pre.data.csrfToken},body:JSON.stringify(body)});const result=await r.json();const c=await read(domain,'csrf');return {status:r.status,type:result.data?.principalType,csrf:c.data.csrfToken,tokenInBody:!!result.data?.token};}
document.querySelector('#run').onclick=async()=>{try{

    public static void main(String[] args)throws Exception {
        String password=UUID.randomUUID()+"临时浏览器😀",code="browser"+UUID.randomUUID().toString().substring(0,12);
